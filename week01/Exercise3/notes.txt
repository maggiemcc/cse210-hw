# Loops:
    - while
    - do-while
    - for
    - foreach

# While loops:
    string response = "yes";

    while (response == "yes")
    {
        Console.Write("Do you want to continue? ");
        response = Console.ReadLine();
    }

# Do-while loops: This loop works the same as a while loop, with the exception that the body of the loop runs once first, before the check is made for the first time. This means that the body of the loop is guaranteed to run at least once.

    ## Example:
        string response;

        do
        {
            Console.Write("Do you want to continue? ");
            response = Console.ReadLine();
        } while (response == "yes");


# For loop: The condition has three parts, separated by semi-colons. 
    for (int i = 0; i < 10; i++)
    {
        Console.WriteLine(i);
    }


# Foreach loop:
    foreach (string color in colors)
    {
        Console.WriteLine(color);
    }
